#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import rospy
import time
from piper_msgs.msg import PosCmd

def low_pass_filter(target, last, alpha=0.1):
    """一阶低通滤波"""
    return alpha * target + (1 - alpha) * last

def descend_motion():
    rospy.init_node("fixed_xy_descend", anonymous=True)

    pub = rospy.Publisher("/pin_pos_cmd", PosCmd, queue_size=1)
    rospy.sleep(0.5)

    # 固定参数
    y_fixed = 0.0
    z_fixed = 0.12

    x_start = 0.34
    x_end   = 0.34 + 0.13
    step    = 0.003
    rate_hz = 10

    # 姿态：末端水平
    roll  = 0.0
    pitch = 1.57
    yaw   = 0.0

    # ===== 滤波参数 =====
    alpha = 0.1              # 越小越稳（0.05~0.15）
    x_filtered = x_start     # 初始化滤波状态

    rate = rospy.Rate(rate_hz)
    x = x_start

    # 初始点
    cmd = PosCmd()
    cmd.x = x_start
    cmd.y = y_fixed
    cmd.z = z_fixed
    cmd.roll  = roll
    cmd.pitch = pitch
    cmd.yaw   = yaw
    cmd.gripper = 0
    cmd.mode1 = 1
    cmd.mode2 = 0

    pub.publish(cmd)
    time.sleep(1)

    while x <= x_end and not rospy.is_shutdown():
        # ===== 低通滤波 =====
        x_filtered = low_pass_filter(x, x_filtered, alpha)

        cmd = PosCmd()
        cmd.x = x_filtered
        cmd.y = y_fixed
        cmd.z = z_fixed
        cmd.roll  = roll
        cmd.pitch = pitch
        cmd.yaw   = yaw
        cmd.gripper = 0
        cmd.mode1 = 1
        cmd.mode2 = 0

        pub.publish(cmd)

        x += step
        rate.sleep()

if __name__ == "__main__":
    descend_motion()
