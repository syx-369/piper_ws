#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import rospy
from piper_msgs.msg import PosCmd

def descend_motion():
    rospy.init_node("ultra_smooth_move", anonymous=True)
    pub = rospy.Publisher("/pin_pos_cmd", PosCmd, queue_size=1)
    rospy.sleep(0.5)

    # 固定参数
    y = 0.0
    z = 0.12

    x_start = 0.34
    x_end   = 0.47

    rate_hz = 200              # 越高越稳（100~300）
    dt = 1.0 / rate_hz

    v_target = 0.01            # m/s（期望速度）
    v = 0.0                    # 当前速度
    a_max = 0.02               # m/s^2（关键参数）

    roll, pitch, yaw = 0.0, 1.57, 0.0

    rate = rospy.Rate(rate_hz)
    x = x_start

    while x < x_end and not rospy.is_shutdown():
        # ===== 限加速度 =====
        dv = v_target - v
        dv_max = a_max * dt
        dv = max(min(dv, dv_max), -dv_max)
        v += dv

        # ===== 积分位置 =====
        x += v * dt

        cmd = PosCmd()
        cmd.x = x
        cmd.y = y
        cmd.z = z
        cmd.roll  = roll
        cmd.pitch = pitch
        cmd.yaw   = yaw
        cmd.gripper = 0
        cmd.mode1 = 1
        cmd.mode2 = 0

        pub.publish(cmd)
        rate.sleep()

if __name__ == "__main__":
    descend_motion()
